cmake_minimum_required(VERSION 3.21)

if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
  project(LinCityCore LANGUAGES CXX)
  include(CTest)
  set(CMAKE_CXX_EXTENSIONS OFF)
  add_compile_definitions($<$<CONFIG:Debug>:DEBUG>)
endif()

if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)

list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
option(ENABLE_NLS "Enable native language support" ON)

if(NOT TARGET ZLIB::ZLIB)
  find_package(ZLIB 1.0 REQUIRED)
endif()
if(NOT TARGET LibXml2::LibXml2)
  find_package(LibXml2 REQUIRED)
endif()
if(NOT TARGET LibXml++-5.0::LibXml++-5.0)
  find_package(LibXml++-5.0 REQUIRED)
endif()
if(NOT TARGET fmt::fmt)
  find_package(fmt 9.0.0 REQUIRED)
endif()
if(ENABLE_NLS AND NOT Intl_FOUND)
  find_package(Intl)
endif()

add_subdirectory(gettext)
add_subdirectory(util)
add_subdirectory(lincity)

if(BUILD_TESTING)
  add_subdirectory(tests)
  add_subdirectory(parity)
endif()
